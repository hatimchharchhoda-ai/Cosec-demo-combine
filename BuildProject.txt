@echo off
REM =============================================================================
REM  MATPOLL SYSTEM — FULL BUILD SCRIPT
REM  Run from:  D:\COSEC-demo\Installer\MatPollInstaller\
REM
REM  READ BEFORE RUNNING:
REM  Not every step needs to run every time.
REM  Each step is labelled [ALWAYS] or [IF CHANGED].
REM  For a normal backend-only change: run steps 1, 2, 8, 9 only.
REM =============================================================================


REM =============================================================================
REM  STEP 1 — PUBLISH COSEC BACKEND                                    [ALWAYS]
REM  Compiles COSEC-demo.csproj into a self-contained Windows exe + DLLs.
REM  Output goes to D:\COSEC-demo\Publish\Backend\
REM =============================================================================

echo [1/9] Publishing COSEC backend...
dotnet publish .\COSEC-demo.csproj -c Release -r win-x64 --self-contained true -o D:\COSEC-demo\Installer\MatPollInstaller\SourceFiles\COSEC\ 
dotnet publish .\MatPoll.csproj -c Release -r win-x64 --self-contained true -o D:\COSEC-demo\Installer\MatPollInstaller\SourceFiles\MATPOLL\
ng build --configuration production


REM =============================================================================
REM  STEP 4b — RE-HARVEST WITH HEAT                                [IF CHANGED]
REM
REM  ONLY run when a NuGet package was added or removed — meaning a new DLL
REM  appeared in or disappeared from SourceFiles\COSEC\ after step 2.
REM  If only file CONTENTS changed (same filenames), SKIP this step entirely.
REM
REM  The -t HeatTransform.xslt flag automatically adds Win64="yes" to every
REM  generated component so you never edit that manually.
REM
REM  AFTER running heat you MUST check COSECServiceDlls.wxs and remove any
REM  Component + ComponentRef entries for these files if heat included them:
REM      COSEC-demo.exe       (managed in COSECService.wxs)
REM      appsettings.json     (written ONLY by the custom action with encryption)
REM      appsettings.Development.json  (optional — remove if not needed)
REM =============================================================================


REM [IF CHANGED] - uncomment to re-harvest:
REM echo  Re-harvesting COSECServiceDlls.wxs...
     "C:\Program Files (x86)\WiX Toolset v3.14\bin\heat.exe" dir ^
         "SourceFiles\COSEC" ^
         -nologo ^
         -sfrag ^
         -srd ^
         -sreg ^
         -gg ^
         -cg COSECServiceDlls ^
         -dr COSECFOLDER ^
         -var var.COSECSourceDir ^
         -out COSECServiceDlls.wxs 
REM echo [4b] Done. Open COSECServiceDlls.wxs and remove COSEC-demo.exe and appsettings.json


REM ── STEP 6: HEAT MatPoll ───────────────────────────────────────────────
echo [6/9] Harvesting MatPoll files...
mkdir SourceFiles\MATPOLL 2>nul
    "C:\Program Files (x86)\WiX Toolset v3.14\bin\heat.exe" dir "SourceFiles\MATPOLL" ^
        -nologo -sfrag -srd -sreg -gg ^
        -cg MatPollFiles ^
        -dr MATPOLLFOLDER ^
        -var var.SourceDir ^
        -out MatPollFiles.wxs
if errorlevel 1 ( echo FAILED step 6 & pause & exit /b 1 )
REM  AFTER running heat: open MatPollFiles.wxs and REMOVE the Component +
REM  ComponentRef for these files:
REM      appsettings.json (written ONLY by the custom action with encryption)
REM      MatPoll.exe      (managed in MatPollService.wxs to install the Windows Service)


REM =============================================================================
REM  STEP 5 — BUILD ConfigCrypto DLL                               [IF CHANGED]
REM  Run when ConfigCryptoHelper.cs or EncryptedConfigExtensions.cs changed.
REM  Also run on FIRST-TIME SETUP.
REM  Output: ConfigCrypto\bin\x64\Release\net472\ConfigCrypto.dll
REM =============================================================================

REM [IF CHANGED] - uncomment:
REM echo [5/9] Building ConfigCrypto...
     "C:\Program Files\Microsoft Visual Studio\2022\Professional\MSBuild\Current\Bin\MSBuild.exe" ^
        ConfigCrypto\ConfigCrypto.csproj ^
        /p:Configuration=Release ^
        /p:Platform=x64 ^
        /p:TargetFramework=net472 ^
        /t:Restore;Build


REM =============================================================================
REM  STEP 6 — BUILD CustomActions DLL                              [IF CHANGED]
REM  Run when CustomActions.cs changed, or after step 5.
REM  Also run on FIRST-TIME SETUP.
REM  Output: CustomActions\bin\x64\Release\net472\CustomActions.dll
REM =============================================================================

REM [IF CHANGED] - uncomment:
REM echo [6/9] Building CustomActions DLL...
     "C:\Program Files\Microsoft Visual Studio\2022\Professional\MSBuild\Current\Bin\MSBuild.exe" ^
         CustomActions\CustomActions.csproj ^
        /p:Configuration=Release ^
        /p:Platform=x64 ^
        /t:Restore;Build


REM =============================================================================
REM  STEP 7 — WRAP WITH MakeSfxCA                                  [IF CHANGED]
REM  Bundles CustomActions.dll + ConfigCrypto.dll + WiX runtime into one
REM  self-extracting DLL (CustomActions.CA.dll) that the MSI embeds.
REM  Run after step 6. Also run on FIRST-TIME SETUP.
REM
REM  ARGUMENT ORDER (exact, do not reorder):
REM    Arg 1: Output file  → CustomActions.CA.dll  (created here)
REM    Arg 2: SfxCA.dll    → WiX self-extractor stub
REM    Arg 3: Your DLL     → CustomActions.dll
REM    Arg 4: DTF DLL      → Microsoft.Deployment.WindowsInstaller.dll
REM    Arg 5: Crypto DLL   → ConfigCrypto.dll  (NEW — added for encryption)
REM    Arg 6: Config       → CustomAction.config (Fixes BadImageFormatException)
REM =============================================================================


REM [IF CHANGED] - uncomment:
REM echo [7/9] Wrapping with MakeSfxCA...
     "C:\Program Files (x86)\WiX Toolset v3.14\sdk\MakeSfxCA.exe" ^
      "D:\COSEC-demo\Installer\MatPollInstaller\CustomActions\bin\x64\Release\net472\CustomActions.CA.dll" ^
      "C:\Program Files (x86)\WiX Toolset v3.14\sdk\x64\SfxCA.dll" ^
      "D:\COSEC-demo\Installer\MatPollInstaller\CustomActions\bin\x64\Release\net472\CustomActions.dll" ^
      "C:\Program Files (x86)\WiX Toolset v3.14\sdk\Microsoft.Deployment.WindowsInstaller.dll" ^
      "D:\COSEC-demo\Installer\MatPollInstaller\CustomActions\bin\x64\Release\net472\ConfigCrypto.dll" ^
      "D:\COSEC-demo\Installer\MatPollInstaller\CustomActions\CustomAction.config"

REM =============================================================================
REM  STEP 8 — WiX CANDLE (Compile .wxs to .wixobj)                     [ALWAYS]
REM  Compiles every .wxs source file into an intermediate .wixobj file.
REM  Always run this — it is fast and safe to always redo.
REM
REM  PATH CHANGE from original:
REM    OLD: -dCustomActionsDir="CustomActions\bin\Release"
REM    NEW: -dCustomActionsDir="CustomActions\bin\x64\Release\net472"
REM  Reason: msbuild /p:Platform=x64 writes output to bin\x64\Release\net472\
REM =============================================================================

echo [8/9] Running WiX Candle...
"C:\Program Files (x86)\WiX Toolset v3.14\bin\candle.exe" -arch x64 ^
    -dSourceDir="SourceFiles\MATPOLL" ^
    -dCOSECSourceDir="SourceFiles\COSEC" ^
    -dCOSECWebSourceDir="SourceFiles\COSECWeb" ^
    -dCustomActionsDir="CustomActions\bin\x64\Release\net472" ^
    Product.wxs ^
    DatabaseSettingsDlg.wxs ^
    MyLicenseDlg.wxs ^
    CustomActionsRef.wxs ^
    COSECService.wxs ^
    COSECServiceDlls.wxs ^
    COSECWeb.wxs ^
    IISSetup.wxs ^
    -ext WixIIsExtension ^
    -ext WixUtilExtension ^
    -ext WixUIExtension


REM =============================================================================
REM  STEP 9 — WiX LIGHT (Link .wixobj to .msi)                         [ALWAYS]
REM  Links all .wixobj files into the final MatPollSystem.msi installer.
REM  Always run after candle.
REM =============================================================================

echo [9/9] Running WiX Light...
"C:\Program Files (x86)\WiX Toolset v3.14\bin\light.exe" ^
    Product.wixobj ^
    DatabaseSettingsDlg.wixobj ^
    MyLicenseDlg.wixobj ^
    CustomActionsRef.wixobj ^
    COSECService.wixobj ^
    COSECServiceDlls.wixobj ^
    COSECWeb.wixobj ^
    IISSetup.wixobj ^
    -ext WixIIsExtension ^
    -ext WixUtilExtension ^
    -ext WixUIExtension ^
    -out MatPollSystem.msi


echo.
echo ============================================================
echo  Build complete.  Output: MatPollSystem.msi
echo ============================================================
echo.
